                                <span style="font-size: 0.75rem; color: #10b981; font-weight: bold;">${item.price}</span>
                            </div>
                            <button onclick="promptDelete(${index})" style="background: none; border: none; color: #ef4444; cursor: pointer; font-size: 0.9rem;" title="ডিলিট"><i class="fa-solid fa-trash"></i></button>
                        </div>
                    `;
                });
                listContainer.innerHTML = listHtml;
            }
            document.getElementById('cartModal').style.display = 'flex';
        }

        function promptDelete(index) {
            itemIndexToDelete = index;
            document.getElementById('deleteConfirmModal').style.display = 'flex';
        }

        function executeDelete() {
            if (itemIndexToDelete !== null) {
                cart.splice(itemIndexToDelete, 1);
                updateCartBadge();
                itemIndexToDelete = null;
            }
            closeModal('deleteConfirmModal');
            openCartModal();
        }

        function checkoutCart() {
            if (cart.length === 0) { alert('কার্ট খালি!'); return; }
            closeModal('cartModal');
            openOrderModal('কালেক্টিভ অর্ডার (' + cart.length + 'টি পণ্য)', 'একাধিক পণ্য');
        }

        db.ref('shopData').once('value').then((snapshot) => {
            try {
                const data = snapshot.val();
                if (!data) {
                    errorBox.style.display = 'block';
                    errorBox.innerHTML = `<i class="fa-solid fa-triangle-exclamation"></i> ডেটা পাওয়া যায়নি।`;
                    renderPlaceholders();
                    hideLoading();
                    return;
                }

                if (data.title) { siteTitleElem.innerText = data.title; siteTitleElem.style.display = 'block'; }
                if (data.bio) { siteBioElem.innerText = data.bio; siteBioElem.style.display = 'block'; }
                if (data.logo) { logoContainer.innerHTML = `<img src="${data.logo}" alt="Logo">`; }

                if (data.company) {
                    let compHtml = '<div class="footer-container">';
                    let hasInfo = false;
                    if (data.company.address) { hasInfo = true; compHtml += `<div class="footer-card"><div class="footer-icon-box"><i class="fa-solid fa-location-dot"></i></div><div class="footer-info"><h4>ঠিকানা</h4><p>${data.company.address}</p></div></div>`; }
                    if (data.company.phone) { hasInfo = true; compHtml += `<div class="footer-card"><div class="footer-icon-box"><i class="fa-solid fa-phone"></i></div><div class="footer-info"><h4>হটলাইন</h4><p>${data.company.phone}</p></div></div>`; }
                    if (data.company.email) { hasInfo = true; compHtml += `<div class="footer-card"><div class="footer-icon-box"><i class="fa-solid fa-envelope"></i></div><div class="footer-info"><h4>ইমেইল</h4><p>${data.company.email}</p></div></div>`; }
                    if (data.company.social) { hasInfo = true; compHtml += `<div class="footer-card"><div class="footer-icon-box"><i class="fa-solid fa-globe"></i></div><div class="footer-info"><h4>সোশ্যাল</h4><p><a href="${data.company.social}" target="_blank">ভিজিট করুন</a></p></div></div>`; }
                    compHtml += '</div><div class="footer-bottom">&copy; ২০২৬ Online Shop. সর্বস্বত্ব সংরক্ষিত।</div>';

                    if (hasInfo) { footerContentText.innerHTML = compHtml; shopFooter.style.display = 'block'; }
                }

                if (data.products) {
                    globalProducts = Object.keys(data.products).map(key => ({ id: key, ...data.products[key] }));
                    displayProducts();
                } else {
                    renderPlaceholders();
                }
                hideLoading();

            } catch (err) {
                hideLoading();
                renderPlaceholders();
            }
        }).catch(() => {
            hideLoading();
            renderPlaceholders();
        });

            document.getElementById('orderModal').style.display = 'flex';
        }
